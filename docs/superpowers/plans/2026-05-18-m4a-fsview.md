```rust
    /// Stat an inode WITHOUT reading file content: `is_dir` from the inode
    /// mode, `size` = the com.apple.decmpfs uncompressed_size if the file is
    /// decmpfs-compressed, else the data stream size (0 if none).
    pub fn stat<D: BlockDevice>(
        &self,
        dev: &mut D,
        inode_num: u64,
    ) -> Result<InodeStat, ContainerError> {
        use crate::inode::Inode;
        use crate::jkey::APFS_TYPE_INODE;

        let mut inode_val: Option<Vec<u8>> = None;
        let root = self.root.clone();
        self.for_each_leaf(dev, &root, 0, &mut |k, v| {
            if inode_val.is_none() {
                let jk = JKey::parse(k)?;
                if jk.obj_type == APFS_TYPE_INODE && jk.obj_id == inode_num {
                    inode_val = Some(v.to_vec());
                }
            }
            Ok(())
        })?;
        let inode = Inode::parse(&inode_val.ok_or_else(oor)?)?;
        let is_dir = inode.mode & 0o170000 == 0o040000; // S_IFMT / S_IFDIR
        let size = match self.get_xattr(dev, inode_num, "com.apple.decmpfs")? {
            Some(d) => crate::decmpfs::DecmpfsHeader::parse(&d)
                .map(|h| h.uncompressed_size)
                .unwrap_or(0),
            None => inode.dstream.as_ref().map(|s| s.size).unwrap_or(0),
        };
        Ok(InodeStat { size, is_dir })
    }
```

Note: `self.root`, `self.for_each_leaf`, `oor()`, `JKey` are already in scope in catalog.rs (used by `read_file_decompressed`). `crate::decmpfs::DecmpfsHeader` + `Inode` are existing pub items.

```rust
#![cfg_attr(test, allow(clippy::unwrap_used, clippy::expect_used, clippy::panic))]
//! apfs — safe high-level filesystem API over `apfs-core`. `FsView` is the
//! OS-agnostic layer a mount host (winfsp-host / FUSE) delegates to.
use std::collections::HashMap;

use apfs_core::block_device::BlockDevice;
use apfs_core::catalog::{Catalog, DirEntry, InodeStat};
use apfs_core::container::{Container, ContainerError};
use apfs_core::jkey::ROOT_DIR_INO_NUM;

/// Errors surfaced by the high-level filesystem view.
#[derive(Debug, thiserror::Error)]
pub enum FsError {
    #[error("apfs-core: {0:?}")]
    Container(#[from] ContainerError),
    #[error("no user volume in container")]
    NoVolume,
}

/// Lightweight attributes for a path (mount-host getattr).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Attr {
    pub inode: u64,
    pub size: u64,
    pub is_dir: bool,
}

/// A read-only filesystem view of the first user volume of an APFS container.
/// Owns the block device; memoizes path→inode and inode→listing (the volume
/// is read-only, so caches never need invalidation).
pub struct FsView<D: BlockDevice> {
    dev: D,
    catalog: Catalog,
    names_hashed: bool,
    path_cache: HashMap<String, u64>,
    dir_cache: HashMap<u64, Vec<DirEntry>>,
}

/// Normalize a path into non-empty components (leading/trailing/duplicate
/// slashes ignored). "/" or "" → no components (the root directory).
fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|c| !c.is_empty()).collect()
}

impl<D: BlockDevice> FsView<D> {
    /// Open the first user volume of the APFS container on `dev`.
    pub fn open(mut dev: D) -> Result<Self, FsError> {
        let c = Container::open(&mut dev)?;
        let bsz = c.superblock.block_size;
        let vols = c.list_user_volumes(&mut dev)?;
        let vol = vols.into_iter().next().ok_or(FsError::NoVolume)?;
        let names_hashed = vol.names_are_hashed();
        let catalog = Catalog::open(&mut dev, &vol, bsz)?;
        Ok(Self {
            dev,
            catalog,
            names_hashed,
            path_cache: HashMap::new(),
            dir_cache: HashMap::new(),
        })
    }

    fn listing(&mut self, ino: u64) -> Result<&[DirEntry], FsError> {
        if !self.dir_cache.contains_key(&ino) {
            let v = self
                .catalog
                .list_dir(&mut self.dev, ino, self.names_hashed)?;
            self.dir_cache.insert(ino, v);
        }
        Ok(self.dir_cache.get(&ino).map(Vec::as_slice).unwrap_or(&[]))
    }

    /// Resolve a path to its inode number (cached). Empty/"/" → root inode.
    pub fn lookup(&mut self, path: &str) -> Result<Option<u64>, FsError> {
        let comps = components(path);
        let key = format!("/{}", comps.join("/"));
        if let Some(&i) = self.path_cache.get(&key) {
            return Ok(Some(i));
        }
        let mut ino = ROOT_DIR_INO_NUM;
        for comp in &comps {
            let next = self
                .listing(ino)?
                .iter()
                .find(|e| e.name == *comp)
                .map(|e| e.file_id);
            match next {
                Some(i) => ino = i,
                None => return Ok(None),
            }
        }
        self.path_cache.insert(key, ino);
        Ok(Some(ino))
    }

    /// List a directory's entries (cached).
    pub fn read_dir(&mut self, path: &str) -> Result<Vec<DirEntry>, FsError> {
        match self.lookup(path)? {
            Some(ino) => Ok(self.listing(ino)?.to_vec()),
            None => Ok(Vec::new()),
        }
    }

    /// Stat a path (size + is_dir + inode). `Ok(None)` if the path is absent.
    pub fn getattr(&mut self, path: &str) -> Result<Option<Attr>, FsError> {
        match self.lookup(path)? {
            None => Ok(None),
            Some(inode) => {
                let InodeStat { size, is_dir } = self.catalog.stat(&mut self.dev, inode)?;
                Ok(Some(Attr { inode, size, is_dir }))
            }
        }
    }

    /// Read up to `len` bytes of a regular file at `path` from byte `offset`.
    /// Transparently decompresses decmpfs. `Ok(vec![])` past EOF / absent.
    pub fn read(&mut self, path: &str, offset: u64, len: usize) -> Result<Vec<u8>, FsError> {
        let ino = match self.lookup(path)? {
            Some(i) => i,
            None => return Ok(Vec::new()),
        };
        let content = self.catalog.read_file_decompressed(&mut self.dev, ino)?;
        let off = offset as usize;
        if off >= content.len() {
            return Ok(Vec::new());
        }
        let end = off.saturating_add(len).min(content.len());
        Ok(content.get(off..end).unwrap_or(&[]).to_vec())
    }
}
```

```rust
//! FsView proven against real APFS images: the synthetic apfs-large.img
//! fixture (deterministic) and, if present, the real-USB snapshot
//! /tmp/apfs-rt.img (best-effort, skipped when absent).
#![allow(clippy::unwrap_used, clippy::expect_used, clippy::indexing_slicing)]
use apfs::FsView;
use apfs_core::block_device::FileBlockDevice;
use std::collections::BTreeSet;
use std::path::Path;

#[test]
fn fsview_apfs_large_fixture() {
    let img = concat!(
        env!("CARGO_MANIFEST_DIR"),
        "/../../tests/fixtures/apfs-large.img"
    );
    if !Path::new(img).exists() {
        eprintln!("skip: fixture missing — run `cargo run -p xtask -- gen-fixture --large`");
        return;
    }
    let dev = FileBlockDevice::open(img).expect("open fixture");
    let mut fs = FsView::open(dev).expect("open FsView");

    // root listing
    let root: BTreeSet<String> = fs
        .read_dir("/")
        .expect("read_dir /")
        .into_iter()
        .map(|e| e.name)
        .collect();
    if !root.contains("dir00") {
        eprintln!("skip: size-only fixture — regenerate gen-fixture --large");
        return;
    }
    for d in 0..16 {
        assert!(root.contains(&format!("dir{d:02}")), "root missing dir{d:02}");
    }

    // dir listing + getattr
    let kids: BTreeSet<String> = fs
        .read_dir("/dir00")
        .expect("read_dir /dir00")
        .into_iter()
        .map(|e| e.name)
        .collect();
    for f in 0..64 {
        assert!(kids.contains(&format!("f{f:03}.bin")), "dir00 missing f{f:03}.bin");
    }
    let da = fs.getattr("/dir00").expect("getattr dir00").expect("dir00 exists");
    assert!(da.is_dir, "dir00 is a directory");
    let fa = fs
        .getattr("/dir00/f000.bin")
        .expect("getattr f000")
        .expect("f000 exists");
    assert!(!fa.is_dir);
    assert_eq!(fa.size, 4096);

    // full + offset/len reads
    let full = fs.read("/dir00/f000.bin", 0, 4096).expect("read full");
    assert_eq!(full.len(), 4096);
    assert!(full.iter().all(|&b| b == b'A'));
    let mid = fs.read("/dir00/f000.bin", 100, 50).expect("read mid");
    assert_eq!(mid.len(), 50);
    assert!(mid.iter().all(|&b| b == b'A'));
    assert!(fs.read("/dir00/f000.bin", 99999, 10).expect("past eof").is_empty());

    // absent + root + cache consistency
    assert!(fs.lookup("/nope/x").expect("lookup absent").is_none());
    assert!(fs.getattr("/nope").expect("getattr absent").is_none());
    assert_eq!(fs.lookup("").expect("lookup root"), Some(2));
    let again: BTreeSet<String> = fs
        .read_dir("/dir00")
        .expect("read_dir /dir00 again (cached)")
        .into_iter()
        .map(|e| e.name)
        .collect();
    assert_eq!(again, kids, "cached listing must match");
}
```
